        [Test]
        public void ThresholdCondition_GeneratedDispatchReadsTypedBlackboardValue()
        {
            var owner = CreateOwner();
            try
            {
                using (var request = new ThresholdRequestBuffers(7u, 6u))
                {
                    var entered = Execute(owner, request.Views(BurstCallbackPhase.Enter));
                    AssertSuccessful(entered);

                    var below = Execute(owner, request.Views(BurstCallbackPhase.Tick));
                    AssertSuccessful(below);
                    Assert.That(below.Status, Is.EqualTo(NodeStatus.Failure));

                    request.SetCurrent(8u);
                    var reached = Execute(owner, request.Views(BurstCallbackPhase.Tick));
                    AssertSuccessful(reached);
                    Assert.That(reached.Status, Is.EqualTo(NodeStatus.Success));
                }
            }
            finally
            {
                DisposeOwner(owner);
            }
        }

        [Test]
        public void AsyncWriteAction_GeneratedDispatchRunsThenConsumesCompletion()
        {
            var owner = CreateOwner();
            try
            {
                using (var request = new AsyncRequestBuffers(includeCompletion: true))
                {
                    var entered = Execute(owner, request.Views(BurstCallbackPhase.Enter));
                    AssertSuccessful(entered);

                    var running = Execute(owner, request.Views(BurstCallbackPhase.Tick));
                    AssertSuccessful(running);
                    Assert.That(running.Status, Is.EqualTo(NodeStatus.Running));
                    Assert.That(request.ReadDestination(), Is.EqualTo(SourceValue));
                    Assert.That(request.MemoryStarted, Is.True);
                    Assert.That(request.MemoryValue, Is.EqualTo(SourceValue));
                    Assert.That(request.MemoryOperation, Is.EqualTo(request.ExpectedOperation));
                    Assert.That(running.Transaction.CommandCount, Is.EqualTo(2u));
                    Assert.That(running.Transaction.OperationCount, Is.EqualTo(1u));
                    AssertCommand(request, 0u, NativeBurstDispatchCommandKindV2.Effect, SourceValue);
                    AssertCommand(request, 1u, NativeBurstDispatchCommandKindV2.Start, SourceValue);

                    var completed = Execute(owner, request.Views(BurstCallbackPhase.Tick));
                    AssertSuccessful(completed);
                    Assert.That(completed.Status, Is.EqualTo(NodeStatus.Success));
                    Assert.That(request.ReadDestination(), Is.EqualTo(CompletionValue));
                    Assert.That(request.MemoryValue, Is.EqualTo(CompletionValue));
                    Assert.That(request.CompletionState,
                        Is.EqualTo(NativeBurstDispatchCompletionStateV2.Consumed));
                    Assert.That(completed.Transaction.CommandCount, Is.EqualTo(2u));
                }
            }
            finally
            {
                DisposeOwner(owner);
            }
        }

        [Test]
        public void AsyncWriteAction_GeneratedDispatchAbortPublishesCancellation()
        {
            var owner = CreateOwner();
            try
            {
                using (var request = new AsyncRequestBuffers(includeCompletion: false))
                {
                    AssertSuccessful(Execute(owner, request.Views(BurstCallbackPhase.Enter)));

                    var running = Execute(owner, request.Views(BurstCallbackPhase.Tick));
                    AssertSuccessful(running);
                    Assert.That(running.Status, Is.EqualTo(NodeStatus.Running));
                    Assert.That(request.ReadDestination(), Is.EqualTo(SourceValue));

                    var aborted = Execute(owner, request.Views(BurstCallbackPhase.Abort));
                    AssertSuccessful(aborted);
                    Assert.That(aborted.Transaction.CommandCount, Is.EqualTo(3u));
                    Assert.That(aborted.Transaction.OperationCount, Is.EqualTo(1u));
                    AssertCommand(request, 2u, NativeBurstDispatchCommandKindV2.Cancel, SourceValue);
                    Assert.That(request.OperationState,
                        Is.EqualTo(NativeBurstDispatchOperationStateV2.Tombstoned));
                }
            }
            finally
            {
                DisposeOwner(owner);
            }
        }

        private static NativeBurstDispatchWorkspaceOwnerV2 CreateOwner()
        {
            using (var shape = new CatalogShapeBuffers())
            {
                var value = shape.Value;
                var capacity = new NativeBurstDispatchWorkspaceCapacityV2(
                    32u,
                    new NativeBurstDispatchBindingCapacityV2(
                        8u,
                        32u,
                        8u,
                        64u,
                        4u,
                        FirstOperationSequence));
                Assert.That(NativeBurstDispatchWorkspaceOwnerV2.TryCreate(
                    in value,
                    in capacity,
                    Allocator.Persistent,
                    out var owner,
                    out var failure), Is.True, failure.ToString());
                return owner;
            }
        }

        private static NativeBurstDispatchWorkspaceResultV2 Execute(
            NativeBurstDispatchWorkspaceOwnerV2 owner,
            NativeBurstDispatchWorkspaceRequestViewsV2 views)
        {
            Assert.That(owner.TryBeginRequest(
                in views,
                out var lease,
                out var failure), Is.True, failure.ToString());
            Assert.That(owner.TryAcquireImmediateBatch(
                in lease,
                out var batch,
                out failure), Is.True, failure.ToString());

            var execution = PublicBurstNodeCatalog.ExecuteImmediate(ref batch);

            Assert.That(execution.Code, Is.EqualTo(BurstExecutionCode.Success));
            Assert.That(owner.TryConsumeResult(
                in lease,
                out var result,
                out failure), Is.True, failure.ToString());
            Assert.That(owner.TryReset(in lease, out failure), Is.True, failure.ToString());
            return result;
        }

        private static void AssertSuccessful(NativeBurstDispatchWorkspaceResultV2 result)
        {
            Assert.That(result.Execution.Code, Is.EqualTo(BurstExecutionCode.Success));
            Assert.That(result.CallbackFailure, Is.EqualTo(BurstContextResult.Success));
            Assert.That(result.FrameAcquired, Is.True);
        }

        private static void AssertCommand(
            AsyncRequestBuffers request,
            uint ordinal,
            NativeBurstDispatchCommandKindV2 expectedKind,
            int expectedPayload)
        {
            var command = request.Commands[(int)ordinal];
            Assert.That(command.Kind, Is.EqualTo(expectedKind));
            Assert.That(command.PayloadSize, Is.EqualTo(4u));
            Assert.That(ReadInt32(request.CommandPayloadBytes, (int)command.PayloadOffset),
                Is.EqualTo(expectedPayload));
            if (expectedKind != NativeBurstDispatchCommandKindV2.Effect)
                Assert.That(command.OperationId, Is.EqualTo(request.ExpectedOperation));
        }

        private static void DisposeOwner(NativeBurstDispatchWorkspaceOwnerV2 owner)
        {
            if (owner != null)
                Assert.That(owner.TryDispose(out var failure), Is.True, failure.ToString());
        }

            private readonly NativeArray<byte> _completionPayloadBytes;
            private readonly NativeArray<NativeBurstDispatchOperationV2> _operations;
            private readonly NativeArray<NativeBurstDispatchTransactionControlV2> _transactionControl;

            internal AsyncRequestBuffers(bool includeCompletion)
            {
                _configurationBytes = Array<byte>(20, Allocator.Persistent);
                _memoryBytes = Array<byte>(32, Allocator.Persistent);
                _randomStates = Array<ulong>(0, Allocator.Persistent);
                _randomIncrements = Array<ulong>(0, Allocator.Persistent);
                _resolvedBindings = Array<NativeBurstDispatchResolvedBindingV2>(
                    5, Allocator.Persistent);
                _liveValueBytes = Array<byte>(8, Allocator.Persistent);
                _completions = Array<NativeBurstDispatchCompletionV2>(
                    includeCompletion ? 1 : 0, Allocator.Persistent);
                _completionPayloadBytes = Array<byte>(
                    includeCompletion ? 4 : 0, Allocator.Persistent);
                Commands = Array<NativeBurstDispatchCommandV2>(8, Allocator.Persistent);
                CommandPayloadBytes = Array<byte>(64, Allocator.Persistent);
                _operations = Array<NativeBurstDispatchOperationV2>(4, Allocator.Persistent);
                _transactionControl = Array<NativeBurstDispatchTransactionControlV2>(
                    1, Allocator.Persistent);

                for (var index = 0; index < 5; index++)
                    WriteUInt32(_configurationBytes, index * 4, (uint)index);
                WriteInt32(_liveValueBytes, 0, SourceValue);
                WriteInt32(_liveValueBytes, 4, 0);
                _resolvedBindings[0] = new NativeBurstDispatchResolvedBindingV2(
                    0u, 700u, NativeBurstDispatchBindingV2.NoOffset);
                _resolvedBindings[1] = new NativeBurstDispatchResolvedBindingV2(1u, 701u, 4u);
                _resolvedBindings[2] = new NativeBurstDispatchResolvedBindingV2(
                    2u, 702u, NativeBurstDispatchBindingV2.NoOffset);
                _resolvedBindings[3] = new NativeBurstDispatchResolvedBindingV2(
                    3u, 703u, NativeBurstDispatchBindingV2.NoOffset);
                _resolvedBindings[4] = new NativeBurstDispatchResolvedBindingV2(4u, 704u, 0u);

                ExpectedOperation = new OperationId(
                    new TreeInstanceId(19UL),
                    new RuntimeNodeIndex(RuntimeNodeIndex),
                    ActivationGeneration,
                    FirstOperationSequence);
                if (includeCompletion)
                {
                    _completions[0] = new NativeBurstDispatchCompletionV2(
                        700u,
                        ExpectedOperation,
                        BurstCompletionOutcome.Succeeded,
                        0u);
                    WriteInt32(_completionPayloadBytes, 0, CompletionValue);
                }

                var transaction = new NativeBurstDispatchTransactionControlV2
                {
                    LedgerToken = 0xa1b72020UL,
                    TreeInstanceId = new TreeInstanceId(19UL),
                    NextOperationSequence = FirstOperationSequence
                };

                _transactionControl[0] = transaction;

        private static BurstCatalogHandshake GeneratedHandshake()
        {
            var catalogType = typeof(PublicBurstNodeCatalog);
            return new BurstCatalogHandshake(
                2u,
                PublicBurstNodeCatalog.Fingerprint,
                GeneratedHash(catalogType, "NodeRegistryFingerprint"),
                1u,
                1u,
                GeneratedHash(catalogType, "ConfigurationLayoutFingerprint"),
                GeneratedHash(catalogType, "MemoryLayoutFingerprint"),
                GeneratedHash(catalogType, "AccessLayoutFingerprint"));
        }

        private static BurstHash256 GeneratedHash(Type catalogType, string propertyName)
        {
            var property = catalogType.GetProperty(
                propertyName,
                BindingFlags.Static | BindingFlags.NonPublic);
            Assert.That(property, Is.Not.Null, "Generated catalog fingerprint is missing: " + propertyName);
            return (BurstHash256)property.GetValue(null);
        }

        private static NativeArray<T> Array<T>(int length, Allocator allocator) where T : struct
            => new NativeArray<T>(length, allocator, NativeArrayOptions.ClearMemory);

        private static OperationId ReadOperationId(NativeArray<byte> bytes, int offset)
            => new OperationId(
                new TreeInstanceId(ReadUInt64(bytes, offset)),
                new RuntimeNodeIndex(ReadUInt32(bytes, offset + 8)),
                ReadUInt32(bytes, offset + 12),
                ReadUInt64(bytes, offset + 16));

        private static int ReadInt32(NativeArray<byte> bytes, int offset)
            => unchecked((int)ReadUInt32(bytes, offset));

        private static uint ReadUInt32(NativeArray<byte> bytes, int offset)
            => bytes[offset]
                | (uint)bytes[offset + 1] << 8
                | (uint)bytes[offset + 2] << 16
                | (uint)bytes[offset + 3] << 24;

        private static ulong ReadUInt64(NativeArray<byte> bytes, int offset)
            => ReadUInt32(bytes, offset) | (ulong)ReadUInt32(bytes, offset + 4) << 32;

        private static void WriteInt32(NativeArray<byte> bytes, int offset, int value)
            => WriteUInt32(bytes, offset, unchecked((uint)value));

        private static void WriteUInt32(NativeArray<byte> bytes, int offset, uint value)
        {
            bytes[offset] = (byte)value;
            bytes[offset + 1] = (byte)(value >> 8);
            bytes[offset + 2] = (byte)(value >> 16);
            bytes[offset + 3] = (byte)(value >> 24);
        }
